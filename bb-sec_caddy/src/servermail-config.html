## Maddy Mail Server — declarative config (nix-generated)
## Source: ~/git/cloud/a_solutions/aa-sui_tools-maddy/src/

# ── Base variables
$(hostname) = mail.diegonmarcos.com
$(primary_domain) = diegonmarcos.com
$(local_domains) = $(primary_domain)

tls file /data/tls/fullchain.pem /data/tls/privkey.pem

# ── Local storage & authentication
auth.pass_table local_authdb {
    table sql_table {
        driver sqlite3
        dsn credentials.db
        table_name passwords
    }
}

storage.imapsql local_mailboxes {
    driver sqlite3
    dsn imapsql.db
}

hostname $(hostname)

# ── SMTP (port 25) — inbound from http-to-smtp-proxy-api / CF Worker
smtp tcp://0.0.0.0:25 {
    limits { all rate 20 1s; all concurrency 10 }
    dmarc no
    check { dkim; spf }
    # Local domains must use Submission
    source $(local_domains) { reject 501 5.1.8 }
    default_source {
        destination postmaster $(local_domains) { deliver_to &local_routing }
        default_destination { reject 550 5.1.1 }
    }
}

# ── Submission (ports 465/587) — authenticated outbound
submission tls://0.0.0.0:465 tcp://0.0.0.0:587 {
    limits { all rate 50 1s }
    auth &local_authdb
    source $(local_domains) {
        check { authorize_sender { ... } }
        destination postmaster $(local_domains) { deliver_to &local_routing }
        default_destination { modify { dkim ... }; deliver_to &remote_queue }
    }
}

# ── Outbound relay (OCI SMTP)
target.smtp outbound_delivery {
    targets tcp://[OCI_RELAY]:587
    auth plain [CREDENTIALS]
    starttls yes
}

# ── IMAP access
imap tls://0.0.0.0:993 tcp://0.0.0.0:143 {
    auth &local_authdb
    storage &local_mailboxes
}
